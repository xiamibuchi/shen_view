  set(VIEWER_JPEG_LIBRARY "${CMAKE_BINARY_DIR}/viewer-jpeg/Release/jpeg-static.lib")

  set(VIEWER_JPEG_LIBRARY "${CMAKE_BINARY_DIR}/viewer-jpeg/libjpeg.a")

file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/viewer-jpeg")
ExternalProject_Add(viewer_jpeg_build
  SOURCE_DIR "${viewer_jpeg_SOURCE_DIR}"
  BINARY_DIR "${CMAKE_BINARY_DIR}/viewer-jpeg"
  DOWNLOAD_COMMAND ""
  CMAKE_ARGS -DCMAKE_POLICY_VERSION_MINIMUM=3.5 -DENABLE_SHARED=OFF -DENABLE_STATIC=ON -DWITH_CRT_DLL=ON -DWITH_SIMD=OFF -DWITH_TURBOJPEG=OFF -DCMAKE_BUILD_TYPE=Release
  BUILD_COMMAND ${CMAKE_COMMAND} --build <BINARY_DIR> --config Release --target jpeg-static --parallel 4
  INSTALL_COMMAND ""
  BUILD_BYPRODUCTS "${VIEWER_JPEG_LIBRARY}")
add_library(viewer_jpeg_static STATIC IMPORTED GLOBAL)
set_target_properties(viewer_jpeg_static PROPERTIES IMPORTED_LOCATION "${VIEWER_JPEG_LIBRARY}")
add_dependencies(viewer_jpeg_static viewer_jpeg_build)
set(JPEG_INCLUDE_DIRS "${viewer_jpeg_SOURCE_DIR}" "${viewer_jpeg_SOURCE_DIR}/src" "${CMAKE_BINARY_DIR}/viewer-jpeg")
set(JPEG_LIBRARIES viewer_jpeg_static)
set(JPEG_INCLUDE_DIR "${viewer_jpeg_SOURCE_DIR}/src" CACHE PATH "" FORCE)
set(JPEG_LIBRARY viewer_jpeg_static CACHE FILEPATH "" FORCE)
set(JPEG_FOUND TRUE CACHE BOOL "" FORCE)
