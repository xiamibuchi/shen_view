cmake_minimum_required(VERSION 3.22)
project(viewer_hdr_codecs LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_CXX_STANDARD 17)
set(SHENVIEW_COMPILE_JOBS 4 CACHE STRING "Maximum MSVC compiler processes per library")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
option(SHENVIEW_AVIF_ONLY "Build only the AVIF animation helper" OFF)
set(SHENVIEW_HDR_SDK "" CACHE PATH "Prebuilt Windows x64 dependency SDK")
if(SHENVIEW_HDR_SDK)
  if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The HDR SDK requires Windows x64 and MSVC")
  endif()
  find_package(ShenViewHdr CONFIG REQUIRED
    PATHS "${SHENVIEW_HDR_SDK}/lib/cmake/ShenViewHdr" NO_DEFAULT_PATH)
else()
  include("${CMAKE_CURRENT_LIST_DIR}/../../third_party/native-sdk/hdr-dependencies.cmake")
endif()

add_executable(viewer_avif_animation avif_animation_main.cpp)
target_link_libraries(viewer_avif_animation PRIVATE avif viewer_lcms)
if(MSVC)
  target_compile_options(viewer_avif_animation PRIVATE /utf-8)
endif()
if(SHENVIEW_AVIF_ONLY)
  return()
endif()
add_executable(viewer_raw raw_main.cpp)
target_link_libraries(viewer_raw PRIVATE viewer_raw_lib)
add_executable(viewer_gainmap gainmap_main.cpp)
target_compile_definitions(viewer_gainmap PRIVATE VIEWER_HAS_AVIF=1)
target_link_libraries(viewer_gainmap PRIVATE avif)
if(MSVC)
  target_link_libraries(viewer_gainmap PRIVATE uhdr-static)
  target_compile_options(viewer_gainmap PRIVATE /utf-8)
else()
  target_link_libraries(viewer_gainmap PRIVATE uhdr)
endif()
if(NOT SHENVIEW_HDR_SDK)
  target_include_directories(viewer_gainmap PRIVATE "${ultrahdr_SOURCE_DIR}")
endif()
add_executable(viewer_heif_gainmap heif_gainmap_main.cpp)
target_link_libraries(viewer_heif_gainmap PRIVATE viewer_lcms)
add_executable(viewer_color_profile_test EXCLUDE_FROM_ALL color_profile_test.cpp)
target_link_libraries(viewer_color_profile_test PRIVATE viewer_lcms)
add_custom_target(viewer_hdr_helpers DEPENDS viewer_gainmap viewer_heif_gainmap viewer_raw viewer_avif_animation)
if(MSVC)
  target_compile_options(viewer_heif_gainmap PRIVATE /utf-8)
endif()
